package payments
import("context";"testing";"time";"github.com/google/uuid")
// Reproduction tests assert current vulnerable behavior, not the desired behavior.
func TestSecurityReviewReplayIgnoresOwner(t *testing.T){
 h:=newHarness(); in:=prepareIn("base","","0.001",evmSender);owner:=uuid.New();in.Caller=&owner
 first,err:=h.svc.Prepare(context.Background(),in);if err!=nil{t.Fatal(err)}
 in.Caller=nil;replay,err:=h.svc.Prepare(context.Background(),in)
 if err!=nil||replay.Payment.ID!=first.Payment.ID{t.Fatalf("replay failed: %v",err)}
 if _,_,err:=h.svc.Get(context.Background(),nil,first.Payment.ID);err==nil{t.Fatal("expected GET denial")}
 t.Log("CONFIRMED: anonymous prepare replay returns an owner-protected payment denied by GET")
}
